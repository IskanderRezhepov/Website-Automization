                  function f_cust_search(param){
                    $("#buttons").hide();
                    $("#b_dossier_attitude").hide();
                    $("#b_doc_attitude").hide();
                    $("#b_add").hide();
                    $("#cust_table").empty();
                    $("#dossier_info").empty();
                    $("#document_info").empty();
                    $("#cust_load").show();
                    switch(param) {
                         case "search":
                             p_min = 0;
                             p_max = $("#h_count").val();
                             break;
                        case "next":
                            p_min = p_max;
                            p_max = parseInt(p_max) + parseInt($("#h_count").val());
                            break;
                        case "prev":
                            p_max = p_min;
                            p_min = parseInt(p_min) - parseInt($("#h_count").val()); - parseInt($("#h_count").val());
                            break;
                   }
                   $.ajax({
                           url:"pkg_w_e_dossier.p_main_table",
                       //   alert("sds");
                           type:"POST",
                           data:{p_cust_shortname   : $("#t_cust_shortname").val(),
                                 p_dossier_number   : $("#t_dossier_number").val(),
                                 p_iin_bin          : $("#t_iin_bin").val(),
                                 p_pcust_shortname  : $("#t_pcust_shortname").val(),
                                 p_sloan_agreement  : $("#t_sloan_agreement").val(),
                                 p_min_row_to_fetch : p_min,
                                 p_max_row_to_fetch : p_max},
                           success:function(data)
                                   {
                                    $("#cust_load").hide();
                                    $("#cust_table").empty();
                                    $("#cust_table").append(data);
                                   },
                                    error: function(xhr, status, error){
                                                 var errorMessage = xhr.status + ": " + xhr.statusText + ": "+xhr.responseText;
                                                 alert("Веб-сервер вернул ошибку. Попробуйте выполнить еще раз. " + errorMessage);
                                                 $("#cust_load").hide();
                                                 $("#cust_table").empty();
                                            }
                           });
                  }
